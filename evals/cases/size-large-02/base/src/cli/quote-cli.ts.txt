import { loadCatalog } from '../pricing/catalog.js'
import { isPricingError } from '../pricing/errors.js'
import { formatMoney } from '../pricing/money.js'
import { buildQuote, quoteExpiresAt } from '../pricing/quote.js'
import { isRegion, isTierId } from '../pricing/types.js'
import { pricingConfig } from '../config/pricing.js'

interface CliArgs {
  sku: string
  quantity: number
  tier: string
  region: string
  json: boolean
}

function parseArgs(argv: readonly string[]): CliArgs {
  const flags = new Map<string, string>()
  let json = false

  for (let index = 0; index < argv.length; index += 1) {
    const token = argv[index]
    if (token === '--json') {
      json = true
      continue
    }
    if (!token.startsWith('--')) continue
    const name = token.slice(2)
    const value = argv[index + 1]
    if (value === undefined || value.startsWith('--')) {
      throw new Error(`--${name} requires a value`)
    }
    flags.set(name, value)
    index += 1
  }

  const sku = flags.get('sku')
  const quantity = Number(flags.get('quantity'))
  const tier = flags.get('tier')
  const region = flags.get('region')

  if (!sku) throw new Error('--sku is required')
  if (!Number.isFinite(quantity)) throw new Error('--quantity must be a number')
  if (!tier) throw new Error('--tier is required')
  if (!region) throw new Error('--region is required')

  return { sku, quantity, tier, region, json }
}

function printTable(args: CliArgs, quote: ReturnType<typeof buildQuote>, expiresAt: string): void {
  const rows: Array<[string, string]> = [
    ['sku', args.sku],
    ['quantity', String(args.quantity)],
    ['subtotal', formatMoney(quote.subtotalMinor, quote.currency)],
    ['discount', formatMoney(quote.discountMinor, quote.currency)],
    ['tax', formatMoney(quote.taxMinor, quote.currency)],
    ['total', formatMoney(quote.totalMinor, quote.currency)],
    ['expires', expiresAt],
  ]
  const width = Math.max(...rows.map(([label]) => label.length))
  for (const [label, value] of rows) {
    process.stdout.write(`${label.padEnd(width)}  ${value}\n`)
  }
}

export function run(argv: readonly string[]): void {
  const args = parseArgs(argv)

  if (!isRegion(args.region)) throw new Error(`unsupported region: ${args.region}`)
  if (!isTierId(args.tier)) throw new Error(`unsupported tier: ${args.tier}`)

  const catalog = loadCatalog(
    pricingConfig.defaultCurrency,
    pricingConfig.catalogRevision,
  )

  const request = {
    customerId: 'cli',
    sku: args.sku,
    quantity: args.quantity,
    tier: args.tier,
    region: args.region,
    requestedAt: new Date().toISOString(),
  }

  try {
    const quote = buildQuote(request, catalog)
    const expiresAt = quoteExpiresAt(request)

    if (args.json) {
      process.stdout.write(`${JSON.stringify({ quote, expiresAt })}\n`)
    } else {
      printTable(args, quote, expiresAt)
    }
  } catch (cause) {
    const message = isPricingError(cause) ? cause.message : String(cause)
    process.stderr.write(`quote-cli: ${message}\n`)
    process.exit(1)
  }
}

run(process.argv.slice(2))
