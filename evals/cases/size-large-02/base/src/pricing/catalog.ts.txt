export function loadCatalog(currency: CurrencyCode, revision: string): Catalog {
  return { currency, revision, entries: ENTRIES }
}

export function entryFor(catalog: Catalog, sku: string): CatalogEntry {
  const entry = catalog.entries.find((candidate) => candidate.sku === sku)
  if (!entry) {
    throw new UnknownSkuError(sku)
  }
  if (!entry.active) {
    throw new InactiveSkuError(sku)
  }
  return entry
}

export function skusFor(catalog: Catalog, unit: string): string[] {
  return catalog.entries
    .filter((entry) => entry.active && entry.meteredUnit === unit)
    .map((entry) => entry.sku)
}
