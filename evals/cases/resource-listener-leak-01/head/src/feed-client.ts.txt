import { logger } from './logger.js'
import type { Quote } from './types.js'

const HEARTBEAT_FRAME = JSON.stringify({ op: 'ping' })
const HEARTBEAT_INTERVAL_MS = 15_000
const MIN_RETRY_MS = 500
const MAX_RETRY_MS = 30_000

export interface FeedOptions {
  url: string
  symbols: readonly string[]
  onQuote: (quote: Quote) => void
}

export class FeedClient {
  private socket: WebSocket | null = null
  private heartbeat: ReturnType<typeof setInterval> | null = null
  private retry: ReturnType<typeof setTimeout> | null = null
  private attempt = 0
  private closing = false

  constructor(private readonly options: FeedOptions) {}

  start(): void {
    this.closing = false
    this.connect()
  }

  private connect(): void {
    const socket = new WebSocket(this.options.url)
    const lifetime = new AbortController()
    this.socket = socket

    socket.addEventListener(
      'open',
      () => {
        this.attempt = 0
        socket.send(
          JSON.stringify({ op: 'subscribe', symbols: this.options.symbols }),
        )
        this.heartbeat = setInterval(() => {
          socket.send(HEARTBEAT_FRAME)
        }, HEARTBEAT_INTERVAL_MS)
      },
      { signal: lifetime.signal },
    )

    socket.addEventListener(
      'message',
      (event) => {
        this.options.onQuote(JSON.parse(String(event.data)) as Quote)
      },
      { signal: lifetime.signal },
    )

    socket.addEventListener(
      'close',
      () => {
        lifetime.abort()
        this.socket = null
        this.scheduleReconnect()
      },
      { signal: lifetime.signal },
    )
  }

  private scheduleReconnect(): void {
    if (this.closing) return

    const delay = Math.min(MIN_RETRY_MS * 2 ** this.attempt, MAX_RETRY_MS)
    this.attempt += 1
    logger.warn(`feed socket closed, retrying in ${delay}ms`)
    this.retry = setTimeout(() => {
      this.connect()
    }, delay)
  }

  stop(): void {
    this.closing = true
    if (this.retry !== null) {
      clearTimeout(this.retry)
      this.retry = null
    }
    if (this.heartbeat !== null) {
      clearInterval(this.heartbeat)
      this.heartbeat = null
    }
    this.socket?.close()
    this.socket = null
  }
}
