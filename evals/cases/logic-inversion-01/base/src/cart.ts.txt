import { SHIPPING_FEE } from './config.js'

export interface CartItem {
  sku: string
  unitPrice: number
  quantity: number
}

export function subtotal(items: readonly CartItem[]): number {
  return items.reduce((sum, item) => sum + item.unitPrice * item.quantity, 0)
}

export function shippingFee(): number {
  return SHIPPING_FEE
}

export function total(items: readonly CartItem[]): number {
  return subtotal(items) + shippingFee()
}
